### 2.2 Firestore rules (20 min, then deploy)

Add this block to `firestore.rules` before the closing brace of the `database/{db}/documents` match block:

```
// ── referralSuggestions ──
// Bonterra warm-handoff pattern: agency_admins suggest endorsements to
// CRMC without bypassing the gateway authority. CRMC reviews each
// suggestion, accepts (which fires EndorseModal pre-filled) or declines.
match /referralSuggestions/{suggestionId} {
  allow read: if isAuth() && (
    isAdmin() ||
    (isAgency() && resource.data.fromAgencyId == userAgencyId()) ||
    (isAgency() && resource.data.toAgencyId   == userAgencyId())
  );

  allow create: if isAuth() && isAgency() &&
    request.resource.data.fromAgencyId == userAgencyId() &&
    request.resource.data.fromUserId   == uid() &&
    request.resource.data.status == 'pending' &&
    request.resource.data.reason is string &&
    request.resource.data.reason.size() <= 1000;

  allow update: if isAdmin() &&
    request.resource.data.status in ['accepted', 'declined'];

  allow delete: if false;
}
```

```jsx
import { useState, useEffect } from 'react'
import { collection, addDoc, getDocs, query, orderBy, serverTimestamp } from 'firebase/firestore'
import { db } from '../../firebase'
import { useAuth } from '../../contexts/AuthContext'
import { MdClose, MdSend } from 'react-icons/md'
import toast from 'react-hot-toast'

const URGENCY_OPTIONS = [
  { value: 'low',    label: 'Low — Patient can wait'                     },
  { value: 'medium', label: 'Medium — Should be reviewed this week'      },
  { value: 'high',   label: 'High — Urgent, needs same-day attention'    },
]

export default function SuggestEndorsementModal({ app, request, siblings = [], onClose }) {
  const { user } = useAuth()
  const [agencies, setAgencies] = useState([])
  const [toAgencyId, setToAgencyId] = useState('')
  const [reason, setReason] = useState('')
  const [recommendedAmount, setRecommendedAmount] = useState('')
  const [urgency, setUrgency] = useState('medium')
  const [saving, setSaving] = useState(false)

  useEffect(() => {
    (async () => {
      const snap = await getDocs(query(collection(db, 'agencies'), orderBy('name', 'asc')))
      const alreadyOn = new Set(siblings.map(s => s.agencyId))
      const list = snap.docs
        .map(d => ({ id: d.id, ...d.data() }))
        .filter(a => a.enabled !== false && !alreadyOn.has(a.id))
      setAgencies(list)
    })()
  }, [siblings])

  const handleSubmit = async () => {
    if (!toAgencyId)               { toast.error('Pick an agency to suggest.'); return }
    if (reason.trim().length < 10) { toast.error('Reason must be at least 10 characters.'); return }
    setSaving(true)
    try {
      const target = agencies.find(a => a.id === toAgencyId)
      await addDoc(collection(db, 'referralSuggestions'), {
        fromAgencyId: user.agencyId,
        fromUserId:   user.uid,
        fromUserName: user.name,
        toAgencyId,
        toAgencyName: target?.name ?? toAgencyId,
        requestId:    app.requestId,
        patientName:  app.patientName ?? '',
        amountNeeded: Number(request?.amountNeeded) || 0,
        reason:       reason.trim(),
        recommendedAmount: recommendedAmount ? Number(recommendedAmount) : null,
        urgency,
        status:       'pending',
        createdAt:    serverTimestamp(),
        respondedAt:  null,
        respondedBy:  null,
        declineReason: null,
        resultingApplicationId: null,
      })
      toast.success('Suggestion sent to CRMC. They will decide whether to endorse.')
      onClose()
    } catch (err) {
      console.error('[SuggestEndorsement] failed:', err)
      toast.error('Could not send suggestion. Try again.')
    } finally {
      setSaving(false)
    }
  }

  return (
    <div className="fixed inset-0 bg-black/40 z-[200] flex items-center justify-center p-4"
         onClick={e => e.target === e.currentTarget && onClose()}>
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-lg max-h-[90vh] overflow-y-auto">
        <div className="flex items-center justify-between px-5 py-4 border-b border-gray-100">
          <h2 className="text-base font-semibold text-gray-900">Suggest another agency</h2>
          <button onClick={onClose} className="text-gray-400 hover:text-gray-600">
            <MdClose size={20} />
          </button>
        </div>
        <div className="px-5 py-4 space-y-4">
          <p className="text-xs text-gray-500 leading-relaxed">
            CRMC reviews each suggestion and decides whether to endorse the case to the suggested
            agency. Your suggestion does not commit any agency to action.
          </p>

          <div>
            <label className="block text-xs font-medium text-gray-700 mb-1">Suggested agency *</label>
            <select className="input" value={toAgencyId} onChange={e => setToAgencyId(e.target.value)}>
              <option value="">Pick an agency</option>
              {agencies.map(a => <option key={a.id} value={a.id}>{a.name}</option>)}
            </select>
            {agencies.length === 0 && (
              <p className="text-xs text-amber-600 mt-1">All agencies are already on this case.</p>
            )}
          </div>

          <div>
            <label className="block text-xs font-medium text-gray-700 mb-1">Reason *</label>
            <textarea
              className="input resize-none" rows={3} maxLength={1000}
              placeholder="e.g. Patient mentions chemotherapy regime — falls under PCSO MAP scope"
              value={reason} onChange={e => setReason(e.target.value)} />
            <p className="text-xs text-gray-400 mt-0.5">{reason.length}/1000</p>
          </div>

          <div>
            <label className="block text-xs font-medium text-gray-700 mb-1">
              Recommended amount (₱) <span className="text-gray-400 font-normal">— optional</span>
            </label>
            <input type="number" min={0} className="input" placeholder="e.g. 10000"
                   value={recommendedAmount} onChange={e => setRecommendedAmount(e.target.value)} />
          </div>

          <div>
            <label className="block text-xs font-medium text-gray-700 mb-1">Urgency</label>
            <select className="input" value={urgency} onChange={e => setUrgency(e.target.value)}>
              {URGENCY_OPTIONS.map(o => <option key={o.value} value={o.value}>{o.label}</option>)}
            </select>
          </div>
        </div>
        <div className="px-5 py-3 border-t border-gray-50 flex gap-2 justify-end">
          <button className="btn-secondary text-sm" onClick={onClose} disabled={saving}>Cancel</button>
          <button className="btn-primary text-sm flex items-center gap-1.5" onClick={handleSubmit}
                  disabled={saving || !toAgencyId || reason.trim().length < 10}>
            <MdSend size={14} /> {saving ? 'Sending…' : 'Send suggestion'}
          </button>
        </div>
      </div>
    </div>
  )
}
```

import Layout from '../../components/Layout'
import { useState, useEffect } from 'react'
import { collection, query, where, orderBy, onSnapshot } from 'firebase/firestore'
import { db } from '../../firebase'
import { useAuth } from '../../contexts/AuthContext'
import { tsToDate } from '../../utils/dates'
import { MdShield, MdPerson, MdBusiness } from 'react-icons/md'

export default function PatientAccessLog() {
  const { user } = useAuth()
  const [entries, setEntries] = useState([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (!user?.uid) return
    const q = query(
      collection(db, 'auditLog'),
      where('patientId', '==', user.uid),
      orderBy('createdAt', 'desc'),
    )
    const unsub = onSnapshot(q,

      () => setLoading(false),
    )
    return unsub
  }, [user?.uid])

## §4 — Failure recovery procedures

### 4.1 Service account expired or wrong

Symptom: `auth/invalid-credential` or `401 Unauthorized`.

1. https://console.firebase.google.com/project/mapa-crmc/settings/serviceaccounts/adminsdk
2. "Generate new private key" → downloads JSON
3. Move to project root:
   ```powershell
   mv ~/Downloads/mapa-crmc-firebase-adminsdk-*.json "C:\Users\MY PC\Downloads\mapa-web\service-account.json"
   ```
4. Verify NOT in git: `git status --short` (should not list `service-account.json`)
5. Re-run §1.2 to verify.

### 4.2 Demo accounts drifted

Symptom: `check-demo-accounts.js` shows `WRONG_ROLE`, `BAD_PASSWORD`, or `NO_PROFILE`.

```powershell
# Verify quota first
# (paste §1.1 probe)

# Repair
node scripts/repair-demo-accounts.js --dry-run
node scripts/repair-demo-accounts.js

# Re-verify
node scripts/check-demo-accounts.js
```

If repair script hangs on Admin SDK writes → use §4.4 REST workaround.

### 4.3 Patient marked for deletion

Symptom: Patient login throws "This account is marked for deletion."

```powershell
node -e "
import('fs').then(async ({ readFileSync }) => {
  const sa = JSON.parse(readFileSync('./service-account.json', 'utf8'));
  const adminApp = await import('firebase-admin/app');
  const { getAuth } = await import('firebase-admin/auth');
  const { JWT } = await import('google-auth-library');
  adminApp.initializeApp({ credential: adminApp.applicationDefault() });
  const uid = (await getAuth().getUserByEmail('patient@gmail.com')).uid;
  const jwt = new JWT({ email: sa.client_email, key: sa.private_key, scopes: ['https://www.googleapis.com/auth/datastore'] });
  const token = (await jwt.getAccessToken()).token;
  const resp = await fetch('https://firestore.googleapis.com/v1/projects/mapa-crmc/databases/(default)/documents/users/' + uid + '?updateMask.fieldPaths=deletion', {
    method: 'PATCH', headers: { Authorization: 'Bearer ' + token, 'Content-Type': 'application/json' },
    body: JSON.stringify({ fields: { deletion: { booleanValue: false } } }),
  });
  console.log('un-mark patient: status=' + resp.status);
});
"
```

### 4.5 Vercel deploy failed

Symptom: live site doesn't reflect pushed changes.

1. Open https://vercel.com/dashboard → `mapa-web`
2. Deployments tab → find latest commit hash
3. If "Error" or "Failed", click into it for the log
4. Most common: TS / build error not caught locally
5. Fix → `npm run build` locally → push again

### 4.6 Firebase rules deploy failed

Symptom: `firebase deploy --only firestore:rules` exits with compile error.

1. Read the error line number
2. Open `firestore.rules`, jump to that line
3. Common errors: missing `;`, mismatched braces, undefined function ref
4. Fix → re-deploy
5. If stuck, revert: `git checkout firestore.rules`

### 4.7 Quota exhausted mid-day

Symptom: writes start returning 429.

1. Stop all write activity immediately
2. Wait until Pacific midnight (3 PM PH)
3. Re-probe with §1.1
4. Use the wait for §3.3 (defense practice)
